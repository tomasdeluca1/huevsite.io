# La guía del MCP, llms.txt y /.well-known (OAuth, AI Catalog, server card)

Allow: /llms.txt
Allow: /llms-full.txt
Allow: /llm.txt
Allow: /.well-known/
# Rutas privadas/internas deshabilitadas.
# Sin la barra final: "Disallow: /dashboard/" NO bloquea /dashboard (el match
# de robots.txt es por prefijo literal), asi que la pagina principal de cada
# seccion privada quedaba crawleable. El prefijo sin barra cubre ambas.
Disallow: /api/
# La server card es metadata pública del MCP (SEP-2127). El resto de /api/
# sigue cerrado. Allow más específico que Disallow, para que gane el match.
Allow: /api/mcp/server-card
Disallow: /_next/
Disallow: /dashboard
Disallow: /onboarding
Disallow: /admin
Disallow: /welcome
Disallow: /checkout
Disallow: /auth/
Disallow: /lab
# NO se bloquean ?from= ni utm_*: esas URLs se comparten en redes y los
# crawlers de preview (facebookexternalhit, LinkedInBot) caen bajo "*", asi
# que bloquearlas romperia el OG de todo link con campaña. Los duplicados ya
# los resuelve el <link rel="canonical"> de cada pagina.
# /login queda crawleable a proposito: tiene OG propio y se comparte.
Allow: /api/og/*

# X / Twitter cards
User-agent: Twitterbot
Allow: /
Allow: /api/
Allow: /api/og/
